cmake_minimum_required(VERSION 3.18)
project(rakmagic_cloud LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CORE "${CMAKE_CURRENT_SOURCE_DIR}/core" CACHE PATH "SA-MP protocol core")
set(R "${CORE}/vendor/raknet")
set(SOURCES BitStream CheckSum DataBlockEncryptor DataCompressor DS_ByteQueue DS_HuffmanEncodingTree DS_Table EncodeClassName GetTime InternalPacketPool LinuxStrings NetworkIDGenerator NetworkTypes PluginInterface RakClient RakNetStatistics RakNetworkFactory RakPeer rakserver RakSleep Rand ReliabilityLayer rijndael RPCMap SHA1 SimpleMutex SocketLayer StringCompressor StringTable SystemAddressList TableSerializer)
foreach(S IN LISTS SOURCES)
  list(APPEND RAK_FILES "${R}/${S}.cpp")
endforeach()
add_executable(rakmagic-worker native/worker.cpp ${RAK_FILES} "${R}/SAMP/SAMPRPC.cpp" "${R}/SAMP/samp_auth.cpp" "${R}/SAMP/samp_netencr.cpp")
target_include_directories(rakmagic-worker PRIVATE native/shim "${CORE}" "${R}" "${R}/SAMP")
target_compile_definitions(rakmagic-worker PRIVATE RAKSAMP_CLIENT _RAKNET_THREADSAFE RAKMAGIC_CLOUD_WG)
target_compile_options(rakmagic-worker PRIVATE -Wno-deprecated-declarations -Wno-write-strings -fno-strict-aliasing -Wno-register)
target_link_libraries(rakmagic-worker PRIVATE pthread)
target_link_options(rakmagic-worker PRIVATE -static-libstdc++ -static-libgcc)
